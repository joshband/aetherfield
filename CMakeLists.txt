cmake_minimum_required(VERSION 3.25)

project(aetherfield LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(cmake/DspSources.cmake)

add_library(aetherfield_dsp STATIC ${AETHERFIELD_DSP_SOURCES})
target_include_directories(aetherfield_dsp PUBLIC src)

add_executable(aetherfield_dsp_tests tests/GainProcessorTests.cpp)
target_link_libraries(aetherfield_dsp_tests PRIVATE aetherfield_dsp)

add_executable(aetherfield_dsp_delay_tests tests/DelayLineTests.cpp)
target_link_libraries(aetherfield_dsp_delay_tests PRIVATE aetherfield_dsp)

add_executable(aetherfield_dsp_fdn_tests tests/FdnTests.cpp)
target_link_libraries(aetherfield_dsp_fdn_tests PRIVATE aetherfield_dsp)

add_executable(aetherfield_dsp_param_tests tests/ParameterTransitionTests.cpp)
target_link_libraries(aetherfield_dsp_param_tests PRIVATE aetherfield_dsp)

add_executable(aetherfield_dsp_allpass_tests tests/SchroederAllpassTests.cpp)
target_link_libraries(aetherfield_dsp_allpass_tests PRIVATE aetherfield_dsp)

add_executable(aetherfield_dsp_diffusion_stereo_tests tests/DiffusionStereoPathTests.cpp)
target_link_libraries(aetherfield_dsp_diffusion_stereo_tests PRIVATE aetherfield_dsp)
target_compile_definitions(aetherfield_dsp PRIVATE AETHERFIELD_TESTING)
target_compile_definitions(aetherfield_dsp_diffusion_stereo_tests PRIVATE AETHERFIELD_TESTING)

add_library(aetherfield_wrapper STATIC
    src/wrapper/ParameterBridge.cpp
)
target_include_directories(aetherfield_wrapper PUBLIC src)
target_link_libraries(aetherfield_wrapper PUBLIC aetherfield_dsp)

add_executable(aetherfield_wrapper_tests tests/ParameterBridgeTests.cpp)
target_link_libraries(aetherfield_wrapper_tests PRIVATE aetherfield_wrapper)

add_executable(aetherfield_render tools/render/main.cpp)
target_link_libraries(aetherfield_render PRIVATE aetherfield_dsp)

add_executable(aetherfield_render_reverb tools/render_reverb/main.cpp)
target_link_libraries(aetherfield_render_reverb PRIVATE aetherfield_dsp)

add_executable(aetherfield_render_diffusion_stereo tools/render_diffusion_stereo/main.cpp)
target_link_libraries(aetherfield_render_diffusion_stereo PRIVATE aetherfield_dsp)

add_executable(aetherfield_render_listening_batch tools/render_listening_batch/main.cpp)
target_link_libraries(aetherfield_render_listening_batch PRIVATE aetherfield_dsp)

if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    foreach(target aetherfield_dsp aetherfield_dsp_tests aetherfield_dsp_delay_tests aetherfield_dsp_fdn_tests aetherfield_dsp_param_tests aetherfield_dsp_allpass_tests aetherfield_dsp_diffusion_stereo_tests aetherfield_wrapper aetherfield_wrapper_tests aetherfield_render aetherfield_render_reverb aetherfield_render_diffusion_stereo aetherfield_render_listening_batch)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endforeach()
endif()

enable_testing()
add_test(NAME aetherfield_dsp_tests COMMAND aetherfield_dsp_tests)
add_test(NAME aetherfield_dsp_delay_tests COMMAND aetherfield_dsp_delay_tests)
add_test(NAME aetherfield_dsp_fdn_tests COMMAND aetherfield_dsp_fdn_tests)
add_test(NAME aetherfield_dsp_param_tests COMMAND aetherfield_dsp_param_tests)
add_test(NAME aetherfield_dsp_allpass_tests COMMAND aetherfield_dsp_allpass_tests)
add_test(NAME aetherfield_dsp_diffusion_stereo_tests COMMAND aetherfield_dsp_diffusion_stereo_tests)
add_test(NAME aetherfield_wrapper_tests COMMAND aetherfield_wrapper_tests)
